Deploy Anywhere

One-line binary install or modern container orchestration.

install.sh 
bash <(curl -Ls https://raw.githubusercontent.com/konstpic/SharX/main/install.sh)
docker-compose.yml 
services:
  sharx-panel:
    image: harbor.sharxconnect.app/sharx/sharx:latest
    ports:
      - "2053:2053"
      - "2096:2096"
    environment:
      - XUI_DB_HOST=postgres
      - XUI_WEB_PORT=2053
